% ---------------------------------------------------------------- Slide 1: methods + performance
\begin{frame}{Model 1: penalized regression --- methods and performance}
\begin{columns}[T,onlytextwidth]
\column{0.45\textwidth}
\small\textbf{Methods}\par\vspace{2pt}
{\footnotesize
\begin{itemize}\setlength\itemsep{1pt}\setlength\parskip{0pt}
  \item \textbf{Models:} LASSO (L1), Ridge (L2), elastic net (L1+L2)
  \item \textbf{Predictor sets:} 15 covariates vs.\ covariates + \alert{12 behaviors}
  \item \textbf{Split:} 80/20 train/test ($2{,}235 / 559$), stratified on outcome
  \item \textbf{Preprocessing} (training data only): median imputation, missing flags for income, PHQ-9 and drinks/week; all predictors standardized
  \item \textbf{Tuning:} penalty strength by 10-fold CV
  \item \textbf{Added value ($\Delta R^2$):} 2,000-sample test bootstrap + repeated 5-fold CV (25 folds)
  \item \textbf{Stability:} LASSO refit on 200 bootstrap samples
\end{itemize}}

\column{0.51\textwidth}
\small\textbf{Performance} \muted{\footnotesize ($R^2$; baseline test $R^2 = 0$)}\par\vspace{4pt}
{\footnotesize
\setlength\tabcolsep{4pt}\begin{tabular}{@{}llccc@{}}
\toprule
Model & Predictors & Test $R^2$ & CV $R^2$ & Test RMSE \\
\midrule
LASSO       & Covariates   & 0.074 & 0.118 & 8.736 \\
            & \textbf{+ Behaviors}  & \textbf{0.079} & \textbf{0.125} & \textbf{8.713} \\[2pt]
Ridge       & Covariates   & 0.072 & 0.116 & 8.750 \\
            & \textbf{+ Behaviors}  & \textbf{0.079} & \textbf{0.123} & \textbf{8.715} \\[2pt]
Elastic net & Covariates   & 0.074 & 0.117 & 8.736 \\
            & \textbf{+ Behaviors}  & \textbf{0.079} & \textbf{0.125} & \textbf{8.713} \\
\bottomrule
\end{tabular}}
\par\vspace{6pt}
{\footnotesize
\begin{itemize}\setlength\itemsep{1pt}\setlength\parskip{0pt}
  \item Behaviors raise $R^2$ by $\approx$\,\textbf{0.008} in CV \\
        \muted{(better in 20--21 of 25 folds)}
  \item Test-set $\Delta R^2 = 0.005$--$0.008$; 95\% CIs include 0 \\
        \muted{(e.g.\ LASSO: $-0.010$ to $0.021$)}
  \item All three penalties perform almost identically
\end{itemize}}
\end{columns}
\end{frame}

\column{0.41\textwidth}
\small\textbf{Key findings}\par\vspace{2pt}
{\footnotesize
\begin{itemize}\setlength\itemsep{2pt}\setlength\parskip{0pt}
  \item \textbf{Starting BMI dominates:} $\sim$6$\times$ larger than any other predictor
  \item \textbf{More weight loss:} \alert{main meal planner} ($\approx$1 pt more), \alert{current smoking}, \alert{vigorous activity}
  \item \textbf{Less weight loss:} more \alert{sedentary time}, higher \alert{PHQ-9}
  \item Moderate activity and sleep: weaker, less consistent
  \item Missing drinks/week linked to less loss \muted{(likely a skip-pattern effect)}
  \item Drinks/week, former smoking, shopping and sharing roles: rarely selected
\end{itemize}}
\vspace{4pt}
{\scriptsize\muted{Associations only: cross-sectional data; behaviors measured at the end of the weight-change period.}}
